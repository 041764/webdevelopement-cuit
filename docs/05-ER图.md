# E-R 图（Mermaid）

```mermaid
erDiagram
  COLLEGE ||--o{ USER : has
  USER ||--o{ LOCAL_CREDENTIAL : authenticates
  USER ||--o{ REFRESH_TOKEN : owns

  USER ||--o{ USER_ROLE : assigned
  ROLE ||--o{ USER_ROLE : contains

  CLASS ||--o{ CLASS_STUDENT : includes
  USER ||--o{ CLASS_STUDENT : joins
  CLASS ||--o{ CLASS_TUTOR : includes
  USER ||--o{ CLASS_TUTOR : tutors

  CLASS ||--o{ ACTIVITY : has
  ACTIVITY ||--o{ ACTIVITY_SIGNUP : has
  USER ||--o{ ACTIVITY_SIGNUP : applies

  USER ||--o{ PLAN : owns_personal
  CLASS ||--o{ PLAN : owns_class
  PLAN ||--o{ PLAN_ITEM : contains
  PLAN_ITEM ||--o{ PLAN_ITEM_PROGRESS : progresses
  USER ||--o{ PLAN_ITEM_PROGRESS : records

  USER ||--o{ EVALUATION : evaluator
  USER ||--o{ EVALUATION : evaluatee
  EVALUATION ||--o{ EVALUATION_DETAIL : details

  COLLEGE {
    integer id PK
    text name UK
    text created_at
  }
  USER {
    integer id PK
    text user_type
    text user_no
    text name
    integer college_id FK
    text status
    text created_at
    text updated_at
  }
  ROLE {
    integer id PK
    text code UK
    text name
  }
  USER_ROLE {
    integer user_id FK
    integer role_id FK
  }
  LOCAL_CREDENTIAL {
    integer id PK
    integer user_id UK_FK
    text client_salt
    text client_hash
    text server_hash
    text created_at
    text updated_at
  }
  REFRESH_TOKEN {
    integer id PK
    integer user_id FK
    text token_hash UK
    text device_id
    text issued_at
    text expires_at
    text revoked_at
  }
  CLASS {
    integer id PK
    text term
    text name
    integer college_id FK
    text created_at
  }
  CLASS_STUDENT {
    integer class_id FK
    integer student_user_id FK
    text joined_at
  }
  CLASS_TUTOR {
    integer class_id FK
    integer tutor_user_id FK
    text assigned_at
  }
  ACTIVITY {
    integer id PK
    integer class_id FK
    text term
    text title
    text description
    integer capacity
    integer requires_review
    text status
    integer created_by_user_id FK
    text created_at
  }
  ACTIVITY_SIGNUP {
    integer id PK
    integer activity_id FK
    integer user_id FK
    text status
    text created_at
    text reviewed_at
    integer reviewed_by_user_id FK
  }
  PLAN {
    integer id PK
    text owner_type
    integer owner_user_id FK
    integer owner_class_id FK
    text term
    text title
    text created_at
  }
  PLAN_ITEM {
    integer id PK
    integer plan_id FK
    text title
    text status
    integer sort_order
    text due_date
    text created_at
    text updated_at
  }
  PLAN_ITEM_PROGRESS {
    integer id PK
    integer plan_item_id FK
    integer percent
    text note
    integer created_by_user_id FK
    text created_at
  }
  EVALUATION {
    integer id PK
    integer evaluator_user_id FK
    integer evaluatee_user_id FK
    text term
    integer score_total
    text comment
    text created_at
  }
  EVALUATION_DETAIL {
    integer id PK
    integer evaluation_id FK
    text item_key
    integer score
    text comment
  }
```
